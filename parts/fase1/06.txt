  if((payCategory==='doubles'||payCategory==='hazmat_doubles') && currentLevel<3){ alert('Doubles/bitrem só pode ser registrado após liberar o Nível 3.'); return; }

  const departureAt=document.getElementById('tripDepartureAt').value;
  const arrivalAt=document.getElementById('tripArrivalAt').value;
  if(!origin || !destination || miles<=0 || !departureAt || !arrivalAt){ alert('Preencha origem, destino, milhas, data/hora de saída e data/hora de chegada.'); return; }
  if(new Date(arrivalAt)<=new Date(departureAt)){ alert('A chegada precisa ser posterior à saída.'); return; }
  const date=departureAt.slice(0,10);
  state.trips=state.trips||[];
  state.trips.push({id:Date.now(),date,departureAt,arrivalAt,origin,originCompany:originCompany||'—',destination,destinationCompany:destinationCompany||'—',cargo:cargo||'—',type,payCategory,miles,week:state.currentWeek});

  document.getElementById('tripOriginCompany').value='';
  document.getElementById('tripDestinationCompany').value='';
  document.getElementById('tripCargo').value='';
  document.getElementById('tripMiles').value='';
  document.getElementById('tripDepartureAt').value='';
  document.getElementById('tripArrivalAt').value='';

  showToast(`Viagem registrada na Semana ${registeredWeek}: ${miles.toLocaleString('pt-BR')} mi.`);
}
function deleteTrip(id){
  const trip=(state.trips||[]).find(t=>t.id===id);
  if(!trip) return;
  if((Number(trip.week)||1)!==state.currentWeek){ alert('Esta viagem pertence a uma semana já fechada e não pode mais ser alterada.'); return; }
  if(!confirm('Excluir esta viagem da semana atual e do total de milhas?')) return;
  state.trips=(state.trips||[]).filter(t=>t.id!==id);
  save();
}
