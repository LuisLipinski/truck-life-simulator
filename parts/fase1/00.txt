      <div class="guide-step"><div class="guide-num">1</div><div><strong>Crie ou selecione uma carreira</strong><p class="muted">Na tela de Carreiras, informe motorista, cidade-base, empresa, dinheiro ao chegar e custos iniciais.       <div class="guide-step"><div class="guide-num">2</div><div><strong>Comece no Nível 1</strong><p class="muted">O Nível 1 é local/regional, com day cab, salário bruto fixo de US$ 850 por semana e meta de 10.000 milhas. Não há per diem nem pagamento por milha.</p></div></div>
      <div class="guide-step"><div class="guide-num">3</div><div><strong>Registre cada deslocamento em Progresso</strong><p class="muted">Informe saída, chegada, origem, destino, empresas, carga, Loaded/Deadhead e milhas.       <div class="guide-step"><div class="guide-num">4</div><div><strong>Registre infrações ou acidentes</strong><p class="muted">Use a aba Infrações e Acidentes quando algo acontecer. Você escolhe se o custo sai imediatamente do saldo ou fica pendente para o próximo holerite.</p></div></div>

Nível 1 — Trainee / Local Driver</summary><ul><li>Salário: <strong>US$ 850 bruto/semana</strong>.</li><li>Não há pagamento por milha no Nível 1; as milhas servem para experiência e progressão.</li><li>Horário padrão: <strong>07:00–15:30</strong>, segunda a sexta, com 30 min de refeição.</li><li>Day cab, sem cama. function parseCSV(text){
  const rows=[];let row=[],cell='',quoted=false;
  for(let i=0;i<text.length;i++){
    const ch=text[i];
    if(quoted){if(ch==='"'&&text[i+1]==='"'){cell+='"';i++;}else if(ch==='"')quoted=false;else cell+=ch;}
    else{if(ch==='"')quoted=true;else if(ch===','){row.push(cell);cell='';}else if(ch==='\n'){row.push(cell);rows.push(row);row=[];cell='';}else if(ch!=='\r')cell+=ch;}
  }
  if(cell.length||row.length){row.push(cell);rows.push(row);}return rows;
}

function perDiemDaysForTrips(trips){
  const valid=new Set();
  for(const t of trips){
    if(!t.departureAt||!t.arrivalAt)continue;
    const start=new Date(t.departureAt),end=new Date(t.arrivalAt);
    if(Number.isNaN(start.getTime())||Number.isNaN(end.getTime())||end<=start)continue;
    const startDay=new Date(start.getFullYear(),start.getMonth(),start.getDate());
    const endDay=new Date(end.getFullYear(),end.getMonth(),end.getDate());
    if(endDay<=startDay)continue;
    for(let d=new Date(startDay);d<=endDay;d.setDate(d.getDate()+1))valid.add(`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`);
  }
  return {days:valid.size,dates:[...valid].sort()};
}

function syncTripDateCompatibility(t){
  if(!t.departureAt&&t.date)t.departureAt=t.date+'T08:00';

}

function estimateTaxes(gross){
  const ss=gross*.062,medicare=gross*.0145,sdi=gross*.013;
  let federal=0;if(gross>300)federal=(Math.min(gross,900)-300)*.10+Math.max(0,gross-900)*.12;federal=Math.max(0,federal);
  let ca=0;if(gross>500)ca=(Math.min(gross,1000)-500)*.02+Math.max(0,gross-1000)*.04;ca=Math.max(0,ca);
  return{ss,medicare,sdi,federal,ca,total:ss+medicare+sdi+federal+ca};
}
function applyPendingIncidents(maxDeduction){
  let available=Math.max(0,maxDeduction),used=0;
  for(const x of state.incidents){
    if(x.payment!=='payslip'||x.status!=='pending'||available<=0)continue;
    const due=Number(x.remaining??x.amount??0);const take=Math.min(due,available);used+=take;available-=take;
    const rem=due-take;if(rem<=.001){x.status='paid';x.remaining=0;}else{x.remaining=rem;}
  }
  return used;
}
function generatePayslip(){
  const trips=currentWeekTrips();if(!trips.length&&!confirm(`A Semana ${state.currentWeek} não possui viagens registradas. Deseja fechar a semana mesmo assim?`))return;

  else{
