function normalizeCitySearch(value){
  return String(value||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().trim();
}
function escapeCityHtml(value){return String(value).replace(/[&<>\"']/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;','\"':'&quot;',"'":'&#39;'}[ch]));}
function cityResultLabel(city,raw){
  const q=normalizeCitySearch(raw); if(!q) return escapeCityHtml(city);
  const normalized=normalizeCitySearch(city),pos=normalized.indexOf(q); if(pos<0) return escapeCityHtml(city);
  return escapeCityHtml(city.slice(0,pos))+'<span class="city-match">'+escapeCityHtml(city.slice(pos,pos+raw.length))+'</span>'+escapeCityHtml(city.slice(pos+raw.length));
}
function setupCityAutocomplete(inputId,resultsId,toggleId){
  const input=document.getElementById(inputId),box=document.getElementById(resultsId),toggle=document.getElementById(toggleId);
  if(!input||!box)return; let active=-1;
  const close=()=>{box.classList.remove('open');box.innerHTML='';input.setAttribute('aria-expanded','false');active=-1;};
  const choose=value=>{input.value=value;input.dataset.cityConfirmed='true';close();input.focus();};
  const draw=(matches,raw,allowManual=false)=>{
    box.innerHTML=''; active=-1;
    const header=document.createElement('div'); header.className='city-results-header';
    header.textContent=raw ? `${matches.length} cidade(s) encontrada(s)` : 'Todas as cidades'; box.appendChild(header);
    matches.forEach(city=>{
      const item=document.createElement('div'); item.className='city-option'; item.setAttribute('role','option');
      item.innerHTML=cityResultLabel(city,raw); item.addEventListener('mousedown',e=>{e.preventDefault();choose(city);}); box.appendChild(item);
    });
    if(allowManual && raw){
      const manual=document.createElement('div'); manual.className='city-option manual'; manual.setAttribute('role','option');
      manual.textContent='＋ Adicionar manualmente: “'+raw+'”'; manual.addEventListener('mousedown',e=>{e.preventDefault();choose(raw);}); box.appendChild(manual);
    }
    box.classList.add('open'); input.setAttribute('aria-expanded','true');
  };
  const search=()=>{
    const raw=input.value.trim(); if(!raw){close();return;}
    const q=normalizeCitySearch(raw);
    const matches=[...new Set(ATS_CITIES.filter(city=>normalizeCitySearch(city).includes(q)))].sort((a,b)=>a.localeCompare(b));
    draw(matches,raw,matches.length===0);
  };
  const showAll=()=>draw([...new Set(ATS_CITIES)].sort((a,b)=>a.localeCompare(b)),'',false);
  input.value='';
  input.addEventListener('input',()=>{input.dataset.cityConfirmed='false';search();});
  toggle?.addEventListener('mousedown',e=>{e.preventDefault();if(box.classList.contains('open'))close();else{input.focus();showAll();}});
  input.addEventListener('keydown',e=>{
    if(e.key==='ArrowDown'&&!box.classList.contains('open')){e.preventDefault();if(input.value.trim())search();else showAll();return;}
    const items=[...box.querySelectorAll('.city-option:not(.empty)')];if(!box.classList.contains('open')||!items.length)return;
    if(e.key==='ArrowDown'){e.preventDefault();active=(active+1)%items.length;}
    else if(e.key==='ArrowUp'){e.preventDefault();active=(active-1+items.length)%items.length;}
    else if(e.key==='Enter'&&active>=0){e.preventDefault();items[active].dispatchEvent(new MouseEvent('mousedown',{bubbles:true}));return;}
    else if(e.key==='Escape'){close();return;}else return;
    items.forEach((el,i)=>el.classList.toggle('active',i===active));items[active]?.scrollIntoView({block:'nearest'});
  });
  input.addEventListener('blur',()=>setTimeout(close,160));
}

  const levelLockedText=document.getElementById('careerLevelLockedText');

  const levelNames={1:'Trainee / Local Driver',2:'Company Driver / OTR',3:'Doubles'};
  document.getElementById('currentLevelDisplay').textContent=`Nível ${state.careerLevel}`;

  const goalLabel = state.careerLevel===1 ? '10.000' : '50.000';

  document.getElementById('mileageBar').style.width=Math.min((state.careerMiles/goal)*100,100)+'%';

  updateOverviewPromotionBanner();

  const hint=document.getElementById('promotionHint');
