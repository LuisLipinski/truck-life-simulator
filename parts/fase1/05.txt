  if(level>=3){

function getWeekPayBreakdown(weekNumber=state.currentWeek){
  const out={normal:0,hazmat:0,doubles:0,hazmat_doubles:0,deadhead:0};
  for(const t of (state.trips||[]).filter(x=>(Number(x.week)||1)===weekNumber)){
    const miles=Number(t.miles)||0;
    if(String(t.type||'Loaded').toLowerCase()==='deadhead') out.deadhead+=miles;
    else { const cat=Object.prototype.hasOwnProperty.call(PAY_RATES,t.payCategory)?t.payCategory:'normal'; out[cat]+=miles; }
  }
  return out;
}
function calculateWeekMileagePay(weekNumber=state.currentWeek){
  const breakdown=getWeekPayBreakdown(weekNumber); let gross=0,totalMiles=0;
  for(const [cat,miles] of Object.entries(breakdown)){gross+=miles*PAY_RATES[cat];totalMiles+=miles;}
  return {breakdown,gross,totalMiles};
}
function localDateKey(d){
  const y=d.getFullYear(),m=String(d.getMonth()+1).padStart(2,'0'),day=String(d.getDate()).padStart(2,'0');
  return `${y}-${m}-${day}`;
}
function getWeekPerDiemDays(weekNumber=state.currentWeek){
  if(Number(state.careerLevel||1)<2) return 0;
  const days=new Set();
  for(const t of (state.trips||[]).filter(x=>(Number(x.week)||1)===weekNumber)){
    if(!t.departureAt||!t.arrivalAt) continue;
    const start=new Date(t.departureAt), end=new Date(t.arrivalAt);
    if(Number.isNaN(start.getTime())||Number.isNaN(end.getTime())||end<=start) continue;
    const startDay=new Date(start.getFullYear(),start.getMonth(),start.getDate());
    const endDay=new Date(end.getFullYear(),end.getMonth(),end.getDate());
    if(endDay<=startDay) continue;
    for(let d=new Date(startDay); d<=endDay; d.setDate(d.getDate()+1)) days.add(localDateKey(d));
  }
  return days.size;
}
function updateAutomaticPerDiem(){
  const el=document.getElementById('perDiemDays'); if(el) el.value=getWeekPerDiemDays(state.currentWeek);
}
function renderWeeklyPayBreakdown(){
  updateAutomaticPerDiem();
  const box=document.getElementById('weeklyPayBreakdown'); if(!box) return;
  const {breakdown,gross,totalMiles}=calculateWeekMileagePay(state.currentWeek);
  const rows=Object.entries(breakdown).filter(([,mi])=>mi>0).map(([cat,mi])=>`${PAY_LABELS[cat]}: <strong>${mi.toLocaleString('pt-BR')} mi × ${money(PAY_RATES[cat])}/mi = ${money(mi*PAY_RATES[cat])}</strong>`);
  box.innerHTML=rows.length?rows.join('<br>')+`<hr style="border-color:var(--line);border-width:1px 0 0;margin:10px 0"><strong>Total: ${totalMiles.toLocaleString('pt-BR')} mi — ${money(gross)} bruto por milhagem</strong>`:'Nenhuma milha registrada na semana atual.';
}

function estimateTaxes(gross){
  const ss=gross*0.062, medicare=gross*0.0145, sdi=gross*0.013;
  let federal=0; if(gross>260) federal=(gross-260)*0.10; if(gross>1000) federal=74+(gross-1000)*0.12;
  let ca=0; if(gross>500) ca=(gross-500)*0.0527;
  return {federal,ss,medicare,sdi,ca};
}

    const hrs=parseFloat(document.getElementById('overrunHours').value)||0;
    const rate=parseFloat(document.getElementById('overrunRate').value)||0;
    gross=base+hrs*rate;
    desc=`Nível 1 — salário semanal${hrs?` + ${hrs}h overrun`:''}`;
  } else {
    const calc=calculateWeekMileagePay(state.currentWeek);
    const miles=calc.totalMiles;
    const days=getWeekPerDiemDays(state.currentWeek);
    document.getElementById('perDiemDays').value=days;
    const pd=parseFloat(document.getElementById('perDiemRate').value)||0;
    gross=calc.gross; perDiem=days*pd;
    const parts=Object.entries(calc.breakdown).filter(([,mi])=>mi>0).map(([cat,mi])=>`${PAY_LABELS[cat]} ${mi.toLocaleString('pt-BR')} mi @ ${money(PAY_RATES[cat])}`);

  }

  const t=estimateTaxes(gross); const taxes=t.federal+t.ss+t.medicare+t.sdi+t.ca;
  const netSalary=gross-taxes-ben; const beforeIncidents=Math.max(0,netSalary+perDiem); const incidentDeduction=applyPendingIncidentDeductions(beforeIncidents); const deposit=beforeIncidents-incidentDeduction;
  document.getElementById('payslip').innerHTML=`
    <table><tbody>
      <tr><td>Salário bruto</td><td>${money(gross)}</td></tr>
      ${level==='1'?'':Object.entries(calculateWeekMileagePay(state.currentWeek).breakdown).filter(([,mi])=>mi>0).map(([cat,mi])=>`<tr><td>${PAY_LABELS[cat]} (${mi.toLocaleString('pt-BR')} mi × ${money(PAY_RATES[cat])})</td><td>${money(mi*PAY_RATES[cat])}</td></tr>`).join('')}
      <tr><td>Federal</td><td>-${money(t.federal)}</td></tr>
      <tr><td>Social Security</td><td>-${money(t.ss)}</td></tr>
      <tr><td>Medicare</td><td>-${money(t.medicare)}</td></tr>
      <tr><td>California Income Tax</td><td>-${money(t.ca)}</td></tr>
      <tr><td>California SDI</td><td>-${money(t.sdi)}</td></tr>
      <tr><td>Benefícios</td><td>-${money(ben)}</td></tr>
      <tr><td><strong>Salário líquido</strong></td><td><strong>${money(netSalary)}</strong></td></tr>
      <tr><td>Per diem</td><td>+${money(perDiem)}</td></tr>
      <tr><td>Infrações/acidentes descontados</td><td>-${money(incidentDeduction)}</td></tr>
      <tr><td><strong>Depósito total</strong></td><td><strong>${money(deposit)}</strong></td></tr>
    </tbody></table>
    <p class="muted">Estimativa de simulação; retenções reais podem variar.</p>`;

  const weekNumber=state.currentWeek;
  const weekTrips=(state.trips||[]).filter(t=>(Number(t.week)||1)===weekNumber);
  const weekMiles=weekTrips.reduce((sum,t)=>sum+(Number(t.miles)||0),0);
  state.closedWeeks=state.closedWeeks||[];
  state.closedWeeks.push({
    week:weekNumber,
    closedAt:new Date().toLocaleString('pt-BR'),
    miles:weekMiles,
    level:Number(level),
    gross, taxes, benefits:ben, netSalary, perDiem, incidentDeduction, deposit, desc
  });
  state.currentWeek=weekNumber+1;
  state.balance+=deposit;
  state.history.push({date:new Date().toLocaleString('pt-BR'),type:'Salário',desc:`Semana ${weekNumber} fechada — ${desc}${incidentDeduction?` — ocorrências: -${money(incidentDeduction)}`:''}`,value:deposit,balance:state.balance});
  save();
  showToast(`Semana ${weekNumber} fechada. Holerite creditado; Semana ${state.currentWeek} iniciada.`);
}
