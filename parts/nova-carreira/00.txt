const KEY='ats_careers_v1';
function load(){try{return JSON.parse(localStorage.getItem(KEY)||'[]')}catch{return []}}
const money=v=>new Intl.NumberFormat('pt-BR',{style:'currency',currency:'USD'}).format(v||0);
const costIds=['costRent','costDeposit','costLicense','costGroceries','costHome','costPhone','costInternet','costTransit'];
const defaultCosts={costRent:1650,costDeposit:1650,costLicense:100,costGroceries:250,costHome:350,costPhone:60,costInternet:75,costTransit:72};
function calc(){
 const arrival=Number(document.getElementById('arrivalBalance').value)||0;
 const total=costIds.reduce((s,id)=>s+(Number(document.getElementById(id).value)||0),0);
 const remaining=arrival-total;
 document.getElementById('arrivalSummary').textContent=money(arrival);
 document.getElementById('costSummary').textContent=money(total);
 const r=document.getElementById('remainingSummary');r.textContent=money(remaining);r.className='remaining '+(remaining<0?'negative':'positive');
 return {arrival,total,remaining};
}
document.getElementById('arrivalBalance').addEventListener('input',calc);
document.querySelectorAll('.setup-cost').forEach(el=>el.addEventListener('input',calc));
document.getElementById('zeroCosts').addEventListener('click',()=>{costIds.forEach(id=>document.getElementById(id).value='0');calc()});
document.getElementById('restoreCosts').addEventListener('click',()=>{costIds.forEach(id=>document.getElementById(id).value=defaultCosts[id]);calc()});
