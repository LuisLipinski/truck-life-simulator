function normalizeCitySearch(value){return String(value||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().trim();}
function escapeHtml(value){return String(value).replace(/[&<>\"']/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;','\"':'&quot;',"'":'&#39;'}[ch]));}
function cityLabel(city,raw){
 const q=normalizeCitySearch(raw); if(!q) return escapeHtml(city);
 const normalized=normalizeCitySearch(city); const pos=normalized.indexOf(q);
 if(pos<0) return escapeHtml(city);
 return escapeHtml(city.slice(0,pos))+'<span class="city-match">'+escapeHtml(city.slice(pos,pos+raw.length))+'</span>'+escapeHtml(city.slice(pos+raw.length));
}
function setupCityAutocomplete(){
 const input=document.getElementById('city'),box=document.getElementById('cityResults'),toggle=document.getElementById('cityToggle');
 if(!input||!box)return; let active=-1;
 const close=()=>{box.classList.remove('open');box.innerHTML='';input.setAttribute('aria-expanded','false');active=-1;};
 const choose=value=>{input.value=value;input.dataset.cityConfirmed='true';close();input.focus();};
 const draw=(matches,raw,allowManual=false)=>{
   box.innerHTML=''; active=-1;
   const header=document.createElement('div'); header.className='city-results-header';
   header.textContent=raw ? `${matches.length} cidade(s) encontrada(s)` : 'Todas as cidades';
   box.appendChild(header);
   matches.forEach(city=>{
     const item=document.createElement('div'); item.className='city-option'; item.setAttribute('role','option');
     item.innerHTML=cityLabel(city,raw); item.addEventListener('mousedown',e=>{e.preventDefault();choose(city);}); box.appendChild(item);
   });
   if(allowManual && raw){
     const manual=document.createElement('div'); manual.className='city-option manual'; manual.setAttribute('role','option');
     manual.textContent='＋ Adicionar manualmente: “'+raw+'”'; manual.addEventListener('mousedown',e=>{e.preventDefault();choose(raw);}); box.appendChild(manual);
   }
   box.classList.add('open'); input.setAttribute('aria-expanded','true');
 };
 const search=()=>{
   const raw=input.value.trim();
   if(!raw){close(); return;}
   const q=normalizeCitySearch(raw);
   const matches=[...new Set(ATS_CITIES.filter(c=>normalizeCitySearch(c).includes(q)))].sort((a,b)=>a.localeCompare(b));
   draw(matches,raw,matches.length===0);
 };

   const matches=[...new Set(ATS_CITIES)].sort((a,b)=>a.localeCompare(b));
   draw(matches,'',false);
 };

 });

}
